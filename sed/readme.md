sed チートシート                                            

    基本構文                                                    
                                                                
    sed [オプション] 'コマンド' ファイル                        

    よく使うオプション

    ┌────────────┬────────────────────────────────┐
    │ オプション │              説明              │
    ├────────────┼────────────────────────────────┤
    │ -i         │ ファイルを直接編集（in-place） │
    ├────────────┼────────────────────────────────┤
    │ -i.bak     │ バックアップ作って直接編集     │
    ├────────────┼────────────────────────────────┤
    │ -n         │ 自動出力を抑制（p と併用）     │
    ├────────────┼────────────────────────────────┤
    │ -e         │ 複数コマンドを指定             │
    ├────────────┼────────────────────────────────┤
    │ -E / -r    │ 拡張正規表現を使用             │
    └────────────┴────────────────────────────────┘

    置換 (s)

    sed 's/old/new/'       # 各行の最初の1つだけ置換
    sed 's/old/new/g'      # 全置換
    sed 's/old/new/gi'     # 大文字小文字無視で全置換
    sed 's/old/new/2'      # 各行の2番目だけ置換

    行指定

    sed '3s/old/new/'      # 3行目だけ置換
    sed '2,5s/old/new/'    # 2〜5行目を置換
    sed '2,$s/old/new/'    # 2行目〜最終行
    sed '/pattern/s/old/new/'  # pattern にマッチする行だけ置換

    削除 (d)

    sed '3d'               # 3行目を削除
    sed '2,5d'             # 2〜5行目を削除
    sed '/pattern/d'       # pattern を含む行を削除
    sed '/^$/d'            # 空行を削除
    sed '/^#/d'            # コメント行を削除

    表示 (p)

    sed -n '3p'            # 3行目だけ表示
    sed -n '2,5p'          # 2〜5行目だけ表示
    sed -n '/pattern/p'    # マッチする行だけ表示（grep 的）

    挿入・追加

    sed '3i\新しい行'       # 3行目の前に挿入 (insert)
    sed '3a\新しい行'       # 3行目の後に追加 (append)
    sed '$a\最終行に追加'   # ファイル末尾に追加

    後方参照

    sed -E 's/(foo)(bar)/\2\1/'    # foobar → barfoo
    sed -E 's/([0-9]+)/[\1]/g'     # 123 → [123]

    実用例

    # ファイル内の全 foo を bar に置換（直接編集）
    sed -i 's/foo/bar/g' file.txt

    # 特定行の間を削除
    sed '10,20d' file.txt

    # 行頭にプレフィックス追加
    sed 's/^/PREFIX: /' file.txt

    # 行末に追加
    sed 's/$/ SUFFIX/' file.txt

    # 区切り文字を変更（/ が含まれるパスに便利）
    sed 's|/usr/local|/opt|g' file.txt

    # 複数コマンド
    sed -e 's/foo/bar/g' -e 's/baz/qux/g' file.txt
